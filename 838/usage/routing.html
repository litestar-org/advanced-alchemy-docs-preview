Read/Write Routing#

Advanced Alchemy routes write operations (INSERT, UPDATE, DELETE, and SELECT ... FOR UPDATE) to a primary engine group while distributing read queries across replica engine groups.

Routing Architecture#

        flowchart LR
    Session["Routing Session"] --> Selector["Operation & Context Check"]
    Selector -->|"Write / FOR UPDATE / Sticky"| Primary["Primary Group (default)"]
    Selector -->|"Read (Clean Context)"| Replica["Read Group (read)"]

  1. Mutations and locks: INSERT, UPDATE, DELETE, and SELECT ... FOR UPDATE statements bind to default_group (primary).

  2. Sticky-after-write consistency: Once a session performs a write, subsequent reads stay on the primary engine until commit() or rollback() resets stickiness.

  3. Load-balanced reads: Clean read operations select an engine from read_group using RoutingStrategy.ROUND_ROBIN or RoutingStrategy.RANDOM.

Routing Configuration#

Configure named engine groups on RoutingConfig and pass it to SQLAlchemyAsyncConfig or SQLAlchemySyncConfig:

from advanced_alchemy.config import SQLAlchemyAsyncConfig
from advanced_alchemy.config.routing import EngineConfig, RoutingConfig, RoutingStrategy

config = SQLAlchemyAsyncConfig(
    routing_config=RoutingConfig(
        engines={
            "default": ["postgresql+asyncpg://user:pass@primary:5432/db"],
            "read": [
                EngineConfig(
                    connection_string="postgresql+asyncpg://user:pass@replica-1:5432/db",
                    weight=2,
                    name="replica-east",
                ),
                EngineConfig(
                    connection_string="postgresql+asyncpg://user:pass@replica-2:5432/db",
                    weight=1,
                    name="replica-west",
                ),
            ],
            "analytics": ["postgresql+asyncpg://user:pass@analytics:5432/db"],
        },
        default_group="default",
        read_group="read",
        routing_strategy=RoutingStrategy.ROUND_ROBIN,
        sticky_after_write=True,
    ),
)

Bind Groups and Context Managers#

Route operations to specific engine groups with use_bind_group(), primary_context(), replica_context(), or the repository bind_group parameter:

from typing import Any

from advanced_alchemy.routing import (
    primary_context,
    replica_context,
    reset_routing_context,
    stick_to_primary_var,
    use_bind_group,
)


async def query_across_groups(repo: Any, item_id: Any) -> tuple[Any, list[Any], int]:
    with primary_context():
        critical_row = await repo.get(item_id)

    with replica_context():
        replica_rows = await repo.get_many()

    with use_bind_group("analytics"):
        analytics_total = await repo.count()

    if stick_to_primary_var.get():
        reset_routing_context()

    return critical_row, replica_rows, analytics_total

You can also pass bind_group="analytics" directly to repository methods such as await repo.get_many(bind_group="analytics").

Complete Runnable Example#

The example below initializes primary and replica SQLite databases, verifies sticky-after-write behavior on the primary engine, and reads committed data from the replica group:

"""Runnable read/write replica routing example for Advanced Alchemy."""

import shutil
import tempfile
from pathlib import Path

from sqlalchemy import create_engine
from sqlalchemy.orm import Mapped

from advanced_alchemy.base import BigIntAuditBase
from advanced_alchemy.config import SQLAlchemySyncConfig
from advanced_alchemy.config.routing import EngineConfig, RoutingConfig, RoutingStrategy
from advanced_alchemy.repository import SQLAlchemySyncRepository
from advanced_alchemy.routing import primary_context, replica_context, reset_routing_context

__all__ = (
    "Article",
    "ArticleRepository",
    "main",
)


class Article(BigIntAuditBase):
    """Article model used to verify read/write replica routing."""

    __tablename__ = "routing_example_article"

    title: Mapped[str]


class ArticleRepository(SQLAlchemySyncRepository[Article]):
    """Synchronous repository for routed Article persistence."""

    model_type = Article


def main() -> None:
    """Execute a complete primary-write and replica-read routing workflow."""
    reset_routing_context()
    with tempfile.TemporaryDirectory() as tmp_dir:
        primary_db = Path(tmp_dir) / "primary.sqlite"
        replica_db = Path(tmp_dir) / "replica.sqlite"
        primary_url = f"sqlite:///{primary_db}"
        replica_url = f"sqlite:///{replica_db}"

        primary_init_engine = create_engine(primary_url)
        Article.metadata.create_all(primary_init_engine)
        primary_init_engine.dispose()
        shutil.copyfile(primary_db, replica_db)

        config = SQLAlchemySyncConfig(
            routing_config=RoutingConfig(
                engines={
                    "default": [EngineConfig(connection_string=primary_url, name="writer")],
                    "read": [EngineConfig(connection_string=replica_url, weight=2, name="reader-1")],
                },
                routing_strategy=RoutingStrategy.ROUND_ROBIN,
                sticky_after_write=True,
            ),
        )

        with config.get_session() as session:
            repo = ArticleRepository(session=session)
            created = repo.add(Article(title="Primary Write"))
            created_id = created.id
            sticky_read = repo.get(created_id)
            if sticky_read.title != "Primary Write":
                msg = f"Unexpected sticky read title: {sticky_read.title}"
                raise RuntimeError(msg)
            session.commit()

        shutil.copyfile(primary_db, replica_db)

        with config.get_session() as session:
            repo = ArticleRepository(session=session)
            with replica_context():
                replica_articles = repo.get_many()
                if len(replica_articles) != 1 or replica_articles[0].title != "Primary Write":
                    msg = f"Unexpected replica articles: {replica_articles}"
                    raise RuntimeError(msg)

            with primary_context():
                primary_article = repo.get_one(title="Primary Write")
                if primary_article.id != created_id:
                    msg = f"Unexpected primary article id: {primary_article.id}"
                    raise RuntimeError(msg)

Disabling Routing for Debugging#

Set enabled=False on RoutingConfig to send all read and write queries to the first engine in default_group:

from advanced_alchemy.config.routing import RoutingConfig

disabled_routing = RoutingConfig(
    engines={
        "default": ["postgresql+asyncpg://user:pass@primary:5432/db"],
        "read": ["postgresql+asyncpg://user:pass@replica:5432/db"],
    },
    enabled=False,
)
