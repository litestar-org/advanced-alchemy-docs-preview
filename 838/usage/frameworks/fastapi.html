FastAPI Integration#

Advanced Alchemy integrates with FastAPI through AdvancedAlchemy, providing engine lifecycle management, request-scoped SQLAlchemy sessions, service and filter dependency generators, and CLI migration commands.

Installation#

Install Advanced Alchemy with FastAPI and an async database driver:

uv add advanced-alchemy fastapi aiosqlite

pip install advanced-alchemy fastapi aiosqlite

pdm add advanced-alchemy fastapi aiosqlite

poetry add advanced-alchemy fastapi aiosqlite

Extension Initialization#

Configure SQLAlchemyAsyncConfig or SQLAlchemySyncConfig and bind AdvancedAlchemy to your FastAPI application:

from fastapi import FastAPI
from advanced_alchemy.extensions.fastapi import (
    AdvancedAlchemy,
    AsyncSessionConfig,
    SQLAlchemyAsyncConfig,
)

app = FastAPI()
alchemy = AdvancedAlchemy(
    config=SQLAlchemyAsyncConfig(
        connection_string="sqlite+aiosqlite:///:memory:",
        session_config=AsyncSessionConfig(expire_on_commit=False),
        create_all=True,
        commit_mode="autocommit",
    ),
    app=app,
)

Models, Schemas, and Services#

Define declarative SQLAlchemy models, Pydantic schemas, and repository services:

class AuthorModel(base.UUIDBase):
    """Author database model for FastAPI example."""

    __tablename__ = "fastapi_basic_author"
    name: Mapped[str]


class AuthorCreate(BaseModel):
    """Request payload for creating an author."""

    name: str


class AuthorSchema(BaseModel):
    """Response schema for an author record."""

    model_config = {"from_attributes": True}

    id: UUID
    name: str


class AuthorRepository(repository.SQLAlchemyAsyncRepository[AuthorModel]):
    """Author async repository."""

    model_type = AuthorModel


class AuthorService(service.SQLAlchemyAsyncRepositoryService[AuthorModel, AuthorRepository]):
    """Author async repository service."""

    repository_type = AuthorRepository

Dependency Injection and Filters#

Use alchemy.provide_session(), alchemy.provide_service(...), and alchemy.provide_filters(...) with FastAPI’s Depends(...) to inject database sessions, services, and OpenAPI-documented query filters (IDs, search strings, boolean flags, enum choices, date ranges, and pagination) into route handlers:

def create_app(db_path: Path) -> FastAPI:
    """Create a FastAPI application wired with AdvancedAlchemy service and filter providers."""
    app = FastAPI()
    alchemy_config = SQLAlchemyAsyncConfig(
        connection_string=f"sqlite+aiosqlite:///{db_path}",
        session_config=AsyncSessionConfig(expire_on_commit=False),
        create_all=True,
        commit_mode="autocommit",
    )
    alchemy = AdvancedAlchemy(config=alchemy_config, app=app)
    router = APIRouter()

    @router.post("/authors", status_code=int(HTTPStatus.CREATED))
    async def create_author(
        data: AuthorCreate,
        authors_service: Annotated[AuthorService, Depends(alchemy.provide_service(AuthorService))],
    ) -> AuthorSchema:
        """Create a new author record."""
        obj = await authors_service.create(data)
        return authors_service.to_schema(obj, schema_type=AuthorSchema)

    @router.get("/authors")
    async def list_authors(
        authors_service: Annotated[AuthorService, Depends(alchemy.provide_service(AuthorService))],
        filters: Annotated[
            list[filters.FilterTypes],
            Depends(
                alchemy.provide_filters(
                    {
                        "id_filter": UUID,
                        "search": "name",
                        "pagination_type": "limit_offset",
                    },
                ),
            ),
        ],
    ) -> service.OffsetPagination[AuthorSchema]:
        """List authors with limit/offset pagination and search filters."""
        results, total = await authors_service.get_many_and_count(*filters)
        return authors_service.to_schema(
            results,
            total,
            filters=filters,
            schema_type=AuthorSchema,
        )

    app.include_router(router)
    return app

When multiple database configurations are registered, pass the target bind_key to alchemy.provide_session("reporting") or alchemy.provide_service(ReportService, key="reporting").

Complete Runnable Example#

The complete runnable example in docs/examples/fastapi/service_di.py exercises service and filter dependency injection in-process using FastAPI’s TestClient:

"""FastAPI service and filter dependency injection example with Advanced Alchemy."""

import tempfile
from http import HTTPStatus
from pathlib import Path
from typing import Annotated
from uuid import UUID

from fastapi import APIRouter, Depends, FastAPI
from fastapi.testclient import TestClient
from pydantic import BaseModel
from sqlalchemy.orm import Mapped

from advanced_alchemy.extensions.fastapi import (
    AdvancedAlchemy,
    AsyncSessionConfig,
    SQLAlchemyAsyncConfig,
    base,
    filters,
    repository,
    service,
)

__all__ = (
    "AuthorCreate",
    "AuthorModel",
    "AuthorRepository",
    "AuthorSchema",
    "AuthorService",
    "create_app",
    "main",
)


class AuthorModel(base.UUIDBase):
    """Author database model for FastAPI example."""

    __tablename__ = "fastapi_basic_author"
    name: Mapped[str]


class AuthorCreate(BaseModel):
    """Request payload for creating an author."""

    name: str


class AuthorSchema(BaseModel):
    """Response schema for an author record."""

    model_config = {"from_attributes": True}

    id: UUID
    name: str


class AuthorRepository(repository.SQLAlchemyAsyncRepository[AuthorModel]):
    """Author async repository."""

    model_type = AuthorModel


class AuthorService(service.SQLAlchemyAsyncRepositoryService[AuthorModel, AuthorRepository]):
    """Author async repository service."""

    repository_type = AuthorRepository


def create_app(db_path: Path) -> FastAPI:
    """Create a FastAPI application wired with AdvancedAlchemy service and filter providers."""
    app = FastAPI()
    alchemy_config = SQLAlchemyAsyncConfig(
        connection_string=f"sqlite+aiosqlite:///{db_path}",
        session_config=AsyncSessionConfig(expire_on_commit=False),
        create_all=True,
        commit_mode="autocommit",
    )
    alchemy = AdvancedAlchemy(config=alchemy_config, app=app)
    router = APIRouter()

    @router.post("/authors", status_code=int(HTTPStatus.CREATED))
    async def create_author(
        data: AuthorCreate,
        authors_service: Annotated[AuthorService, Depends(alchemy.provide_service(AuthorService))],
    ) -> AuthorSchema:
        """Create a new author record."""
        obj = await authors_service.create(data)
        return authors_service.to_schema(obj, schema_type=AuthorSchema)

    @router.get("/authors")
    async def list_authors(
        authors_service: Annotated[AuthorService, Depends(alchemy.provide_service(AuthorService))],
        filters: Annotated[
            list[filters.FilterTypes],
            Depends(
                alchemy.provide_filters(
                    {
                        "id_filter": UUID,
                        "search": "name",
                        "pagination_type": "limit_offset",
                    },
                ),
            ),
        ],
    ) -> service.OffsetPagination[AuthorSchema]:
        """List authors with limit/offset pagination and search filters."""
        results, total = await authors_service.get_many_and_count(*filters)
        return authors_service.to_schema(
            results,
            total,
            filters=filters,
            schema_type=AuthorSchema,
        )

    app.include_router(router)
    return app


def main() -> None:
    """Execute the FastAPI dependency injection example in-process."""
    with tempfile.TemporaryDirectory() as tmp_dir:
        db_file = Path(tmp_dir) / "fastapi_example.sqlite"
        app = create_app(db_file)
        with TestClient(app=app) as client:
            create_res = client.post("/authors", json={"name": "Grace Hopper"})
            if create_res.status_code != HTTPStatus.CREATED:
                msg = f"Unexpected create status: {create_res.status_code}"
                raise RuntimeError(msg)
            created = create_res.json()
            if created["name"] != "Grace Hopper":
                msg = f"Unexpected created author name: {created['name']}"
                raise RuntimeError(msg)

            list_res = client.get("/authors?currentPage=1&pageSize=10")
            if list_res.status_code != HTTPStatus.OK:
                msg = f"Unexpected list status: {list_res.status_code}"
                raise RuntimeError(msg)
            payload = list_res.json()
            if payload["total"] != 1 or payload["items"][0]["name"] != "Grace Hopper":
                msg = f"Unexpected list payload: {payload}"
                raise RuntimeError(msg)
