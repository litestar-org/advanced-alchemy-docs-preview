Litestar Integration#

See also

Litestar’s documentation for SQLAlchemy integration

Advanced Alchemy provides first-class integration with Litestar through SQLAlchemyPlugin, which re-exports core models, repositories, services, filters, and data transfer objects under advanced_alchemy.extensions.litestar.

Installation#

Install Advanced Alchemy with the litestar extra using your preferred package manager:

uv add "advanced-alchemy[litestar]" aiosqlite

pip install "advanced-alchemy[litestar]" aiosqlite

pdm add "advanced-alchemy[litestar]" aiosqlite

poetry add "advanced-alchemy[litestar]" aiosqlite

Plugin and Engine Configuration#

Configure SQLAlchemyAsyncConfig or SQLAlchemySyncConfig and register SQLAlchemyPlugin on your Litestar application. The plugin manages engine lifecycles, request-scoped database sessions, exception handling, and CLI migration commands:

def create_app() -> Litestar:
    """Construct the Litestar application with an in-memory SQLite database."""
    return Litestar(
        route_handlers=[AuthorController],
        plugins=[
            SQLAlchemyPlugin(
                config=SQLAlchemyAsyncConfig(
                    connection_string="sqlite+aiosqlite:///:memory:",
                    create_all=True,
                ),
            ),
        ],
    )

Models, Schemas, and Data Transfer Objects#

Define declarative models using the base classes in advanced_alchemy.base (re-exported as advanced_alchemy.extensions.litestar.base), alongside msgspec.Struct schemas or SQLAlchemyDTO definitions:

class AuthorModel(base.UUIDBase):
    """Author database model."""

    __tablename__ = "litestar_basic_author"
    name: Mapped[str]


class AuthorCreate(msgspec.Struct):
    """Payload for creating an author."""

    name: str


class AuthorSchema(msgspec.Struct):
    """Serialized author schema."""

    id: UUID
    name: str


AuthorDTO = SQLAlchemyDTO[AuthorModel]
AuthorWriteDTO = SQLAlchemyDTO[
    Annotated[
        AuthorModel,
        SQLAlchemyDTOConfig(exclude={"id"}),
    ]
]

SQLAlchemyDTO automatically inspects model columns, relationships, and @property / @cached_property descriptors (exposing properties as read-only fields):

from functools import cached_property
from sqlalchemy.orm import Mapped
from advanced_alchemy.extensions.litestar import SQLAlchemyDTO, base


class UserAccount(base.UUIDAuditBase):
    __tablename__ = "litestar_doc_user_account"
    first_name: Mapped[str]
    last_name: Mapped[str]

    @property
    def full_name(self) -> str:
        return f"{self.first_name} {self.last_name}"

    @cached_property
    def display_length(self) -> int:
        return len(self.full_name)


UserAccountDTO = SQLAlchemyDTO[UserAccount]

Repositories and Services#

Bind a model to a repository and service class to encapsulate query execution, pagination, and schema serialization:

class AuthorRepository(repository.SQLAlchemyAsyncRepository[AuthorModel]):
    """Author async repository."""

    model_type = AuthorModel


class AuthorService(service.SQLAlchemyAsyncRepositoryService[AuthorModel, AuthorRepository]):
    """Author async repository service."""

    repository_type = AuthorRepository

Controllers and Dependency Injection#

Use create_service_dependencies() to wire service instances and query filter dependencies (such as ID filters, search filters, date bounds, and limit/offset pagination) directly into route handlers:

class AuthorController(Controller):
    """Author HTTP endpoints."""

    path = "/authors"
    dependencies = providers.create_service_dependencies(
        AuthorService,
        "authors_service",
        filters={"id_filter": UUID, "search": "name", "pagination_type": "limit_offset"},
    )

    @post(path="/")
    async def create_author(
        self,
        authors_service: NamedDependency[AuthorService],
        data: AuthorCreate,
    ) -> AuthorSchema:
        """Create a new author record."""
        obj = await authors_service.create(data, auto_commit=True)
        return authors_service.to_schema(obj, schema_type=AuthorSchema)

    @get(path="/")
    async def list_authors(
        self,
        authors_service: NamedDependency[AuthorService],
        filters: NamedDependency[SkipValidation[list[filters.FilterTypes]]],
    ) -> service.OffsetPagination[AuthorSchema]:
        """List authors with standard pagination and filtering."""
        results, total = await authors_service.get_many_and_count(*filters)
        return authors_service.to_schema(
            results,
            total,
            filters=filters,
            schema_type=AuthorSchema,
        )

Complete Runnable Example#

The full runnable module is located in docs/examples/litestar/basic_app.py and can be verified in-process with Litestar’s test client:

"""Basic Litestar application with Advanced Alchemy service, DTO, and controller integration."""

from http import HTTPStatus
from typing import Annotated
from uuid import UUID

import msgspec
from litestar import Controller, Litestar, get, post
from litestar.di import NamedDependency
from litestar.params import SkipValidation
from litestar.testing import TestClient
from sqlalchemy.orm import Mapped

from advanced_alchemy.extensions.litestar import (
    SQLAlchemyAsyncConfig,
    SQLAlchemyDTO,
    SQLAlchemyDTOConfig,
    SQLAlchemyPlugin,
    base,
    filters,
    providers,
    repository,
    service,
)

__all__ = (
    "AuthorController",
    "AuthorCreate",
    "AuthorDTO",
    "AuthorModel",
    "AuthorRepository",
    "AuthorSchema",
    "AuthorService",
    "AuthorWriteDTO",
    "create_app",
    "main",
)


class AuthorModel(base.UUIDBase):
    """Author database model."""

    __tablename__ = "litestar_basic_author"
    name: Mapped[str]


class AuthorCreate(msgspec.Struct):
    """Payload for creating an author."""

    name: str


class AuthorSchema(msgspec.Struct):
    """Serialized author schema."""

    id: UUID
    name: str


AuthorDTO = SQLAlchemyDTO[AuthorModel]
AuthorWriteDTO = SQLAlchemyDTO[
    Annotated[
        AuthorModel,
        SQLAlchemyDTOConfig(exclude={"id"}),
    ]
]


class AuthorRepository(repository.SQLAlchemyAsyncRepository[AuthorModel]):
    """Author async repository."""

    model_type = AuthorModel


class AuthorService(service.SQLAlchemyAsyncRepositoryService[AuthorModel, AuthorRepository]):
    """Author async repository service."""

    repository_type = AuthorRepository


class AuthorController(Controller):
    """Author HTTP endpoints."""

    path = "/authors"
    dependencies = providers.create_service_dependencies(
        AuthorService,
        "authors_service",
        filters={"id_filter": UUID, "search": "name", "pagination_type": "limit_offset"},
    )

    @post(path="/")
    async def create_author(
        self,
        authors_service: NamedDependency[AuthorService],
        data: AuthorCreate,
    ) -> AuthorSchema:
        """Create a new author record."""
        obj = await authors_service.create(data, auto_commit=True)
        return authors_service.to_schema(obj, schema_type=AuthorSchema)

    @get(path="/")
    async def list_authors(
        self,
        authors_service: NamedDependency[AuthorService],
        filters: NamedDependency[SkipValidation[list[filters.FilterTypes]]],
    ) -> service.OffsetPagination[AuthorSchema]:
        """List authors with standard pagination and filtering."""
        results, total = await authors_service.get_many_and_count(*filters)
        return authors_service.to_schema(
            results,
            total,
            filters=filters,
            schema_type=AuthorSchema,
        )


def create_app() -> Litestar:
    """Construct the Litestar application with an in-memory SQLite database."""
    return Litestar(
        route_handlers=[AuthorController],
        plugins=[
            SQLAlchemyPlugin(
                config=SQLAlchemyAsyncConfig(
                    connection_string="sqlite+aiosqlite:///:memory:",
                    create_all=True,
                ),
            ),
        ],
    )


def main() -> None:
    """Execute the basic Litestar application example in-process."""
    app = create_app()
    with TestClient(app=app) as client:
        create_response = client.post("/authors/", json={"name": "Ada Lovelace"})
        if create_response.status_code != HTTPStatus.CREATED:
            msg = f"Unexpected create status: {create_response.status_code}"
            raise RuntimeError(msg)
        created = create_response.json()
        if created["name"] != "Ada Lovelace":
            msg = f"Unexpected created author name: {created['name']}"
            raise RuntimeError(msg)

        list_response = client.get("/authors/?currentPage=1&pageSize=10")
        if list_response.status_code != HTTPStatus.OK:
            msg = f"Unexpected list status: {list_response.status_code}"
            raise RuntimeError(msg)
        payload = list_response.json()
        if payload["total"] != 1 or payload["items"][0]["name"] != "Ada Lovelace":
            msg = f"Unexpected list payload: {payload}"
            raise RuntimeError(msg)
